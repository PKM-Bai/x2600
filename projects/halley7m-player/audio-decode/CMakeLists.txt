cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(player_audio_decode_probe LANGUAGES C)
endif()
get_filename_component(PLAYER_DECODE_WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PLAYER_MINIMP3 "${PLAYER_DECODE_WORKSPACE}/third_party/player-minimp3")
set(PLAYER_FAAD2 "${PLAYER_DECODE_WORKSPACE}/third_party/player-faad2")
file(GLOB PLAYER_FAAD_SOURCES CONFIGURE_DEPENDS "${PLAYER_FAAD2}/libfaad/*.c")
add_library(player_faad_fixed STATIC ${PLAYER_FAAD_SOURCES})
target_include_directories(player_faad_fixed PUBLIC "${PLAYER_FAAD2}/include")
target_include_directories(player_faad_fixed PRIVATE "${PLAYER_FAAD2}/libfaad")
target_compile_definitions(player_faad_fixed PRIVATE FIXED_POINT LC_ONLY_DECODER DISABLE_SBR
    HAVE_INTTYPES_H=1 HAVE_MEMCPY=1 HAVE_STRING_H=1 HAVE_STRINGS_H=1
    HAVE_SYS_STAT_H=1 HAVE_SYS_TYPES_H=1 PACKAGE_VERSION="2.11.2")
add_library(player_audio_decode STATIC player_audio_decode.c player_mp3_backend.c player_pcm_convert.c)
target_include_directories(player_audio_decode PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(player_audio_decode PRIVATE "${PLAYER_MINIMP3}")
target_link_libraries(player_audio_decode PUBLIC player_faad_fixed)
foreach(target player_audio_decode player_faad_fixed)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    if(CMAKE_SYSTEM_PROCESSOR STREQUAL "mips")
        target_compile_options(${target} PRIVATE -mips32r2 -mabi=32 -mfp32 -mhard-float
            -EL -mno-abicalls -fno-pic -G0 -ffunction-sections -fdata-sections -fno-common -fstack-usage
            $<$<CONFIG:Release>:-O2>)
    endif()
    if(MSVC)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
endforeach()
if(MSVC)
    target_compile_options(player_audio_decode PRIVATE /W4 /WX)
    # Preserve upstream minimp3 byte narrowing; keep our adapter/converter strict.
    set_source_files_properties(player_mp3_backend.c PROPERTIES COMPILE_OPTIONS /wd4244)
else()
    target_compile_options(player_audio_decode PRIVATE -Wall -Wextra -Werror)
    target_compile_options(player_faad_fixed PRIVATE -fwrapv)
    target_link_libraries(player_faad_fixed PUBLIC m)
endif()
option(PLAYER_AUDIO_DECODE_HOST_TEST "Build host decoder/converter probes" OFF)
if(PLAYER_AUDIO_DECODE_HOST_TEST)
    add_executable(player_audio_decode_test "${PLAYER_DECODE_WORKSPACE}/tests/player_audio_decode_test.c")
    target_link_libraries(player_audio_decode_test PRIVATE player_audio_decode)
    add_executable(player_pcm_convert_test "${PLAYER_DECODE_WORKSPACE}/tests/player_pcm_convert_test.c")
    target_link_libraries(player_pcm_convert_test PRIVATE player_audio_decode)
    foreach(target player_audio_decode_test player_pcm_convert_test)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /UNDEBUG)
            target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        endif()
    endforeach()
endif()
