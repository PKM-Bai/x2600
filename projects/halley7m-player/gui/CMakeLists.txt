cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(player_gui_probe LANGUAGES C)
endif()
get_filename_component(PLAYER_GUI_WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PLAYER_LVGL "${PLAYER_GUI_WORKSPACE}/libbare-cpu/projects/x16xx-halley6/Demos/LVGL_GUIDemo/components/lvgl")
set(PLAYER_GUI_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
option(PLAYER_GUI_HOST_TESTS "Build production GUI headless regression" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${PLAYER_GUI_WORKSPACE}/scripts/prepare-player-gui.py"
    --output "${PLAYER_GUI_GENERATED}" RESULT_VARIABLE PLAYER_GUI_PREPARED)
if(NOT PLAYER_GUI_PREPARED EQUAL 0)
    message(FATAL_ERROR "Player LVGL/font preparation failed")
endif()
file(GLOB_RECURSE PLAYER_LVGL_SOURCES CONFIGURE_DEPENDS "${PLAYER_LVGL}/src/*.c")
add_library(player_lvgl STATIC ${PLAYER_LVGL_SOURCES})
target_include_directories(player_lvgl SYSTEM PUBLIC "${PLAYER_LVGL}")
target_include_directories(player_lvgl PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(player_lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE)
set(PLAYER_GUI_SOURCES player_gui.c player_gui_surface.c player_font.c
    "${CMAKE_CURRENT_SOURCE_DIR}/../common/player_input.c")
if(NOT PLAYER_GUI_HOST_TESTS)
    list(APPEND PLAYER_GUI_SOURCES player_gui_display.c)
endif()
add_library(player_gui STATIC ${PLAYER_GUI_SOURCES})
if(MSVC)
    target_compile_options(player_gui PRIVATE /W4 /WX)
else()
    target_compile_options(player_gui PRIVATE -Wall -Wextra -Werror)
endif()
target_include_directories(player_gui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../common")
target_include_directories(player_gui PRIVATE "${PLAYER_GUI_GENERATED}" "${CMAKE_CURRENT_SOURCE_DIR}/../display")
target_link_libraries(player_gui PUBLIC player_lvgl)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${PLAYER_GUI_WORKSPACE}/scripts/prepare-player-gui.py"
    "${PLAYER_GUI_WORKSPACE}/third_party/ReGBA/source/tools/unifont.hex.gz" "${CMAKE_CURRENT_SOURCE_DIR}/player_gui.c")
foreach(target player_lvgl player_gui)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
        if(target STREQUAL "player_lvgl")
            target_compile_options(${target} PRIVATE /W3)
        endif()
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections -fno-common)
    endif()
    if(CMAKE_SYSTEM_PROCESSOR STREQUAL "mips")
        target_compile_options(${target} PRIVATE -mips32r2 -mabi=32 -mfp32 -mhard-float -EL
            -mno-abicalls -fno-pic -G0 -O2)
    endif()
endforeach()
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "mips")
    target_compile_definitions(player_gui PRIVATE PLAYER_GUI_FREERTOS=1)
    target_include_directories(player_gui SYSTEM PRIVATE
        "${PLAYER_GUI_WORKSPACE}/libbare-cpu/cpu/core/include"
        "${PLAYER_GUI_WORKSPACE}/libbare-cpu/cpu/soc-x2600/include"
        "${PLAYER_GUI_WORKSPACE}/libbare-cpu/Middlewares/FreeRTOS-Kernel/include"
        "${CMAKE_CURRENT_SOURCE_DIR}/../mips/include" "${CMAKE_CURRENT_SOURCE_DIR}/../mips/port")
endif()
if(PLAYER_GUI_HOST_TESTS)
    add_executable(player_gui_test tests/player_gui_test.c)
    target_link_libraries(player_gui_test PRIVATE player_gui)
    add_executable(player_display_test tests/player_display_test.c
        "${CMAKE_CURRENT_SOURCE_DIR}/../display/player_display.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/../display/player_display_format.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/../display/player_rotator.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/../common/player_video.c")
    add_executable(player_display_format_test "${PLAYER_GUI_WORKSPACE}/tests/player_display_format_test.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/../display/player_display_format.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/../common/player_video.c")
    add_executable(player_video_format_resolve_test "${PLAYER_GUI_WORKSPACE}/tests/player_video_format_resolve_test.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/../common/player_video.c")
    foreach(format_target player_display_format_test player_video_format_resolve_test)
        target_include_directories(${format_target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../display"
            "${CMAKE_CURRENT_SOURCE_DIR}/../common")
        if(MSVC)
            target_compile_options(${format_target} PRIVATE /utf-8 /W4 /WX /UNDEBUG)
        else()
            target_compile_options(${format_target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        endif()
    endforeach()
    add_executable(player_rotator_test "${CMAKE_CURRENT_SOURCE_DIR}/../display/tests/player_rotator_test.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/../display/player_rotator.c")
    target_include_directories(player_rotator_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../display")
    if(MSVC)
        target_compile_options(player_rotator_test PRIVATE /W4 /WX /UNDEBUG)
    else()
        target_compile_options(player_rotator_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    target_include_directories(player_display_test PRIVATE tests/display-stubs
        "${CMAKE_CURRENT_SOURCE_DIR}/../display" "${CMAKE_CURRENT_SOURCE_DIR}/../common")
    set(PLAYER_BUTTONS_PLATFORM "${PLAYER_GUI_WORKSPACE}/projects/halley7m-regba/platform")
    add_executable(player_buttons_test "${PLAYER_GUI_WORKSPACE}/tests/buttons_test.c" "${PLAYER_BUTTONS_PLATFORM}/buttons.c")
    add_executable(player_buttons_callback_test tests/player_buttons_callback_test.c "${PLAYER_BUTTONS_PLATFORM}/buttons.c")
    foreach(target player_buttons_test player_buttons_callback_test)
        target_include_directories(${target} PRIVATE "${PLAYER_BUTTONS_PLATFORM}")
        target_compile_definitions(${target} PRIVATE BUTTONS_HOST_TEST=1)
        if(MSVC)
            target_compile_options(${target} PRIVATE /utf-8 /W4 /WX /UNDEBUG)
        else()
            target_compile_options(${target} PRIVATE -UNDEBUG)
        endif()
    endforeach()
    if(MSVC)
        target_compile_options(player_gui_test PRIVATE /utf-8 /W4 /WX /UNDEBUG)
        target_compile_definitions(player_gui_test PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(player_display_test PRIVATE /utf-8 /W4 /WX /UNDEBUG)
        target_compile_definitions(player_display_test PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(player_gui_test PRIVATE -UNDEBUG)
        target_compile_options(player_display_test PRIVATE -UNDEBUG)
    endif()
endif()
