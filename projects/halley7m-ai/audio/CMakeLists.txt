cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(halley7m_ai_audio LANGUAGES C)
    set(AI_AUDIO_STANDALONE ON)
    enable_testing()
endif()

add_library(ai_audio_core STATIC ai_air1103.c ai_resampler.c ai_ptt.c)
target_include_directories(ai_audio_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(ai_audio_core PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(MSVC)
    target_compile_options(ai_audio_core PRIVATE /W4 /WX)
else()
    target_compile_options(ai_audio_core PRIVATE -Wall -Wextra -Werror)
endif()

option(AI_AUDIO_BUILD_TESTS "Build native audio core tests" ${AI_AUDIO_STANDALONE})
if(AI_AUDIO_BUILD_TESTS)
    add_executable(ai_audio_test tests/ai_audio_test.c)
    target_link_libraries(ai_audio_test PRIVATE ai_audio_core)
    if(NOT MSVC)
        target_link_libraries(ai_audio_test PRIVATE m)
        target_compile_options(ai_audio_test PRIVATE -Wall -Wextra -Werror)
    else()
        target_compile_options(ai_audio_test PRIVATE /W4 /WX)
    endif()
    set_target_properties(ai_audio_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    add_test(NAME ai_audio_core COMMAND ai_audio_test)
endif()
