set(AI_NETWORK_DEPS "${WORKSPACE}/build/ai-network-deps" CACHE PATH "Pinned network source directory")
set(AI_LWIP "${AI_NETWORK_DEPS}/lwip/src")
if(NOT EXISTS "${AI_LWIP}/include/lwip/init.h")
    message(FATAL_ERROR "Run scripts/prepare-ai-network.py --download before building network firmware")
endif()
file(GLOB AI_LWIP_CORE CONFIGURE_DEPENDS "${AI_LWIP}/core/*.c" "${AI_LWIP}/core/ipv4/*.c" "${AI_LWIP}/api/*.c")
add_library(ai_lwip STATIC ${AI_LWIP_CORE} "${AI_LWIP}/netif/ethernet.c" "${AI_LWIP}/apps/sntp/sntp.c" sys_arch.c)
target_include_directories(ai_lwip PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${AI_LWIP}/include")
target_include_directories(ai_lwip SYSTEM PUBLIC ${PLATFORM_INCLUDES})
add_library(ai_ethernet STATIC ai_ethernet.c ai_ping.c board_gmac.c board_rng.c board_entropy.c)
target_include_directories(ai_ethernet PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ai_ethernet PUBLIC ai_lwip)
target_compile_options(ai_ethernet PRIVATE -Wall -Wextra -Werror)
