cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(halley7m_ai_cloud LANGUAGES C)
    set(AI_CLOUD_STANDALONE ON)
    enable_testing()
endif()
if(NOT TARGET ai_core)
    add_library(ai_cloud_history STATIC ../core/ai_history.c)
    target_include_directories(ai_cloud_history PUBLIC ../core)
    set_target_properties(ai_cloud_history PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    set(AI_CLOUD_HISTORY_TARGET ai_cloud_history)
else()
    set(AI_CLOUD_HISTORY_TARGET ai_core)
endif()
add_library(ai_cloud_protocol STATIC ai_deepseek.c)
target_include_directories(ai_cloud_protocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ai_cloud_protocol PUBLIC ${AI_CLOUD_HISTORY_TARGET})
set_target_properties(ai_cloud_protocol PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(MSVC)
    target_compile_options(ai_cloud_protocol PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(ai_cloud_protocol PRIVATE -Wall -Wextra -Werror)
endif()
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "mips")
    target_compile_options(ai_cloud_protocol PRIVATE -mips32r2 -mabi=32 -mfp32 -mhard-float -EL -mno-abicalls -fno-pic -G0)
    if(TARGET ai_cloud_history)
        target_compile_options(ai_cloud_history PRIVATE -mips32r2 -mabi=32 -mfp32 -mhard-float -EL -mno-abicalls -fno-pic -G0)
    endif()
    set(AI_CLOUD_STANDALONE OFF)
endif()
option(AI_CLOUD_BUILD_TESTS "Build native cloud protocol tests" ${AI_CLOUD_STANDALONE})
if(AI_CLOUD_BUILD_TESTS)
    add_executable(ai_cloud_test tests/ai_cloud_test.c)
    target_link_libraries(ai_cloud_test PRIVATE ai_cloud_protocol)
    set_target_properties(ai_cloud_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    if(MSVC)
        target_compile_options(ai_cloud_test PRIVATE /W4 /WX /utf-8 /UNDEBUG)
    else()
        target_compile_options(ai_cloud_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    add_test(NAME ai_cloud_protocol COMMAND ai_cloud_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/deepseek-recorded-shape.sse")
endif()
