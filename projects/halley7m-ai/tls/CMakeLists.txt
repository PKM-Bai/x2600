cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ai_tls_standalone LANGUAGES C)
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_C_STANDARD_REQUIRED ON)
    if(CMAKE_SYSTEM_PROCESSOR STREQUAL "mips")
        add_compile_options(-mips32r2 -mabi=32 -mfp32 -mhard-float -EL
            -mno-abicalls -fno-pic -G0 -Os -ffunction-sections -fdata-sections)
    endif()
endif()
option(AI_TLS_BUILD_HOST_TESTS "Build Windows loopback TLS test client" OFF)
if(AI_TLS_BUILD_HOST_TESTS)
    if(NOT WIN32 OR CMAKE_CROSSCOMPILING)
        message(FATAL_ERROR "The current host TLS test uses Windows BCrypt and Winsock")
    endif()
    add_executable(ai_tls_host_test tests/tls_host_test.c)
    target_link_libraries(ai_tls_host_test PRIVATE ai_tls bcrypt ws2_32)
    target_compile_definitions(ai_tls_host_test PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
        target_compile_options(ai_tls_host_test PRIVATE /W4 /WX /UNDEBUG /utf-8)
    endif()
endif()
get_filename_component(AI_WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(AI_MBEDTLS_SOURCE "${AI_WORKSPACE}/build/ai-network-deps/mbedtls" CACHE PATH "Pinned complete Mbed TLS 4.1.1 release")
if(NOT EXISTS "${AI_MBEDTLS_SOURCE}/tf-psa-crypto/core/psa_crypto.c")
    message(FATAL_ERROR "Run scripts/prepare-ai-network.py with the complete pinned release")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(MBEDTLS_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/ai_mbedtls_config.h" CACHE FILEPATH "" FORCE)
set(TF_PSA_CRYPTO_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/ai_crypto_config.h" CACHE FILEPATH "" FORCE)
add_subdirectory("${AI_MBEDTLS_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/mbedtls" EXCLUDE_FROM_ALL)
add_library(ai_tls STATIC ai_tls.c)
target_include_directories(ai_tls PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ai_tls PUBLIC mbedtls mbedx509 tfpsacrypto)
if(MSVC)
    target_compile_options(ai_tls PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(ai_tls PRIVATE -Wall -Wextra -Werror)
endif()
